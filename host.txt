[Unit]

After=syslog.target

        root /var/www/rewardwise.co/html/RewardWiseBackendServer;

        server_name internals.rewardwise.co;


        location / {

          proxy_set_header        Host $host;
          proxy_set_header        X-Real-IP $remote_addr;
          proxy_set_header        X-Forwarded-For $proxy_add_x_forwarded_for;
          proxy_set_header        X-Forwarded-Proto $scheme;

          proxy_pass          http://localhost:8000;

          proxy_redirect      http://localhost:8000 http:/internals.rewardwise.co/;

        }
        location /static/ {
        alias /var/www/rewardwise.co/html/RewardWiseBackendServer/static/;
