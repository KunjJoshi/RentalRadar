[Unit]
Description=Gunicorn Instance to Serve Rental Radar
After=network.target

[Service]
User=root
Group=www-data
WorkingDirectory=/var/www/rentalradar.tech/html/RentalRadar
Environment="PATH=/var/www/rentalradar.tech/html/RentalRadar/venv/bin"
ExecStart=/usr/local/bin/gunicorn --workers 6 --threads 2 --timeout 90 --bind 0.0.0.0:8000 wsgi:app


[Install]
WantedBy=multi-user.target


server{

        root /var/www/rentalradar.tech/html/RentalRadar;
        server_name rentalradar.tech www.rentalradar.tech;

        location / {
              proxy_set_header  Host $host;
              proxy_set_header  X-Real-IP $remote_addr;
              proxy_set_header  X-Forwarded-For $proxy_add_x_forwarded_for;
              proxy_set_header  X-Forwarded-Proto $scheme;

              proxy_pass http://0.0.0.0:8000;
              proxy_read_timeout 90;

              proxy_redirect http://0.0.0.0:8000 http://rentalradar.tech;
            }

        location /static/ {
               alias /var/www/rentalradar.tech/html/RentalRadar/static/;
           }

}
